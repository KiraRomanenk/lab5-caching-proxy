# Запуск сервера
npm start -- -h 127.0.0.1 -p 3000 -c cache

# GET: картинки немає в кеші (http.cat), потім з кешу
curl -s -D - -o cat404.jpg http://127.0.0.1:3000/404
ls -l cache
curl -s -D - -o cat404_2.jpg http://127.0.0.1:3000/404
cmp cat404.jpg cat404_2.jpg

# GET: неіснуючий код
curl -s -D - http://127.0.0.1:3000/999

# PUT: своя картинка, потім GET
curl -s -D - -X PUT --data-binary @my.jpg http://127.0.0.1:3000/404
curl -s -D - -o got.jpg http://127.0.0.1:3000/404
cmp my.jpg got.jpg

# PUT без тіла
curl -s -D - -X PUT http://127.0.0.1:3000/418

# DELETE: видалення, повторне видалення
curl -s -D - -X DELETE http://127.0.0.1:3000/404
ls cache
curl -s -D - -X DELETE http://127.0.0.1:3000/404

# Недозволений метод і некоректний шлях
curl -s -D - -X POST http://127.0.0.1:3000/200
curl -s -D - http://127.0.0.1:3000/abc
